# 文档标注项目技术文档 v2

## 1. 架构设计

### 1.1 更简化的设计原则

| 原则 | v1 | v2（更简） |
|-----|----|----|
| 处理流程 | Pipeline + 4个Processor | 1个Annotator类 |
| 数据传递 | 多级中转 | 单一Document对象 |
| 配置方式 | YAML配置文件 | 构造函数参数 |
| 类的数量 | ~10个 | ~5个 |
| 代码量 | ~1250行 | ~600行 |

### 1.2 整体架构（极简版）

```
┌─────────────────────────────────────────────────────────┐
│                    Annotator                             │
│  ┌───────────────────────────────────────────────────┐  │
│  │  1. parse()    - 解析文档                         │  │
│  │  2. detect()   - 检测元素 (调用OCR)               │  │
│  │  3. extract()  - 提取特征 (调用LLM)               │  │
│  │  4. classify() - 分类布局 (调用LLM)               │  │
│  └───────────────────────────────────────────────────┘  │
└─────────────────────────────────────────────────────────┘
           │                    │
           ▼                    ▼
    ┌──────────┐          ┌──────────┐
    │ OCR接口  │          │ LLM接口  │
    └──────────┘          └──────────┘
```

### 1.3 极简目录结构

```
docs_annotation/
```

## 2. 核心设计

### 2.1 扁平化数据模型

```python
# document.py
from dataclasses import dataclass, field
from typing import Optional, Literal
from enum import Enum

class FileType(str, Enum):
    PDF = "pdf"
    DOC = "doc"
    EXCEL = "excel"
    PPT = "ppt"
    HTML = "html"
    TXT = "txt"

class LayoutType(str, Enum):
    SINGLE = "single"
    DOUBLE = "double"
    MIXED = "mixed"

@dataclass
class Document:
    """文档数据模型 - 扁平化设计"""

    # 基础信息
    doc_id: str
    file_type: FileType
    file_path: str = ""
    page_count: int = 0

    # 解析内容（内部使用）
    _pages: list = field(default_factory=list)
    _text: str = ""
    _images: list = field(default_factory=list)

    # PDF标注字段（扁平化，无嵌套Profile）
    layout: LayoutType = LayoutType.SINGLE
    has_image: bool = False
    has_table: bool = False
    has_formula: bool = False
    has_chart: bool = False
    reading_order_sensitive: Optional[bool] = None

    # 表格特征（只在has_table=True时有意义）
    table_dominant: Optional[bool] = None
    long_table: bool = False
    cross_page_table: bool = False

    # 图表特征（只在has_chart=True时有意义）
    cross_page_chart: bool = False

    def to_dict(self) -> dict:
        """转换为字典"""
        return {
            "doc_id": self.doc_id,
            "file_type": self.file_type.value,
            "layout": self.layout.value,
            "has_image": self.has_image,
            "has_table": self.has_table,
            "has_formula": self.has_formula,
            "has_chart": self.has_chart,
            "reading_order_sensitive": self.reading_order_sensitive,
            "table_dominant": self.table_dominant,
            "long_table": self.long_table,
            "cross_page_table": self.cross_page_table,
            "cross_page_chart": self.cross_page_chart,
        }
```

**设计说明**：
- 单一dataclass，无嵌套结构
- 所有字段平铺，简单直观
- `_` 开头的字段为内部使用，输出时自动忽略

### 2.2 核心标注类

```python
# annotator.py
from pathlib import Path
from typing import Optional, Dict, Any
from document import Document, FileType
from ocr import OCRModel
from llm import LLMModel

class Annotator:
    """文档标注器 - 单一类完成所有标注"""

    def __init__(
        self,
        ocr_model: Optional[OCRModel] = None,
        llm_model: Optional[LLMModel] = None
    ):
        """
        初始化标注器

        Args:
            ocr_model: OCR模型（用于元素检测）
            llm_model: LLM模型（用于特征提取和分类）
        """
        self.ocr_model = ocr_model
        self.llm_model = llm_model

    def annotate(self, file_path: str) -> Document:
        """
        标注文档 - 一行代码完成所有标注

        流程: parse → detect → extract → classify
        """
        doc = self.parse(file_path)
        doc = self.detect(doc)
        doc = self.extract(doc)
        doc = self.classify(doc)
        return doc

    def parse(self, file_path: str) -> Document:
        """步骤1: 解析文档"""
        file_type = self._detect_file_type(file_path)

        doc = Document(
            doc_id=Path(file_path).stem,
            file_type=file_type,
            file_path=file_path
        )

        if file_type == FileType.PDF:
            self._parse_pdf(doc)
        elif file_type == FileType.DOC:
            self._parse_docx(doc)
        # ... 其他类型

        return doc

    def detect(self, doc: Document) -> Document:
        """步骤2: 检测元素（图片/表格/公式/图表）"""
        if not self.ocr_model:
            return doc  # 无OCR模型则跳过

        for page in doc._pages:
            result = self.ocr_model.detect_elements(page)
            doc.has_image = doc.has_image or len(result.get("images", [])) > 0
            doc.has_table = doc.has_table or len(result.get("tables", [])) > 0
            doc.has_formula = doc.has_formula or len(result.get("formulas", [])) > 0
            doc.has_chart = doc.has_chart or len(result.get("charts", [])) > 0

        return doc

    def extract(self, doc: Document) -> Document:
        """步骤3: 提取特征（跨页/长表格等）"""
        if not self.llm_model:
            return doc  # 无LLM模型则跳过

        # 提取表格特征
        if doc.has_table:
            table_features = self._extract_table_features(doc)
            doc.table_dominant = table_features.get("table_dominant")
            doc.long_table = table_features.get("long_table", False)
            doc.cross_page_table = table_features.get("cross_page_table", False)

        # 提取图表特征
        if doc.has_chart:
            doc.cross_page_chart = self._extract_chart_features(doc)

        return doc

    def classify(self, doc: Document) -> Document:
        """步骤4: 分类布局（单页/双页/混合）"""
        if not self.llm_model:
            return doc

        prompt = self._build_layout_prompt(doc)
        result = self.llm_model.classify(
            prompt,
            options=["single", "double", "mixed"]
        )
        doc.layout = LayoutType(result["label"])

        return doc

    # ============ 私有辅助方法 ============

    def _detect_file_type(self, file_path: str) -> FileType:
        """检测文件类型"""
        ext = Path(file_path).suffix.lower()
        mapping = {
            ".pdf": FileType.PDF,
            ".doc": FileType.DOC,
            ".docx": FileType.DOC,
            ".xls": FileType.EXCEL,
            ".xlsx": FileType.EXCEL,
            ".ppt": FileType.PPT,
            ".pptx": FileType.PPT,
            ".html": FileType.HTML,
            ".htm": FileType.HTML,
            ".txt": FileType.TXT,
        }
        return mapping.get(ext, FileType.TXT)

    def _parse_pdf(self, doc: Document) -> None:
        """解析PDF"""
        import pdfplumber

        with pdfplumber.open(doc.file_path) as pdf:
            doc.page_count = len(pdf.pages)
            for page in pdf.pages:
                doc._pages.append(page.to_image().orig_bytes)
                doc._text += page.extract_text() or ""

    def _parse_docx(self, doc: Document) -> None:
        """解析Word"""
        from docx import Document as DocxDocument

        docx = DocxDocument(doc.file_path)
        doc.page_count = len(docx.element.body)
        doc._text = "\n".join([p.text for p in docx.paragraphs])

    def _extract_table_features(self, doc: Document) -> Dict[str, Any]:
        """提取表格特征（使用LLM）"""
        prompt = f"""
        分析以下文档的表格特征：
        - 页数: {doc.page_count}
        - 文本预览: {doc._text[:500]}

        判断：
        1. table_dominant: 表格是否是主要内容（占比>50%）
        2. long_table: 是否有跨3页以上的长表格
        3. cross_page_table: 表格是否跨页显示

        返回JSON格式。
        """
        return self.llm_model.extract(prompt, schema={
            "table_dominant": "bool",
            "long_table": "bool",
            "cross_page_table": "bool"
        })

    def _extract_chart_features(self, doc: Document) -> bool:
        """提取图表特征"""
        # 简单实现：检查图表是否跨越多页
        chart_positions = []  # 从OCR结果中获取
        # ... 实现逻辑
        return False

    def _build_layout_prompt(self, doc: Document) -> str:
        """构建布局分类的Prompt"""
        return f"""
        分析文档布局类型：
        - 页数: {doc.page_count}
        - 包含图片: {doc.has_image}
        - 包含表格: {doc.has_table}
        - 文本预览: {doc._text[:500]}

        判断布局类型：
        - single: 单页布局，每页内容独立
        - double: 双页布局，内容跨页连续
        - mixed: 混合布局

        返回类型标签。
        """
```

**设计说明**：
- 单一类完成所有标注逻辑
- 4个步骤方法清晰分离
- 每个步骤可独立调用（方便调试）
- 自动检测文件类型

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List

class OCRModel(ABC):
    """OCR模型基类"""

    @abstractmethod
    def detect_elements(self, page_image: bytes) -> Dict[str, List[Dict]]:
        """
        检测页面元素

        返回:
        {
            "images": [{"bbox": [x1,y1,x2,y2]}],
            "tables": [{"bbox": [x1,y1,x2,y2]}],
            "formulas": [{"bbox": [x1,y1,x2,y2]}],
            "charts": [{"bbox": [x1,y1,x2,y2]}]
        }
        """
        pass

# 实现示例

    def __init__(self):
```

### 2.4 LLM接口（保持简单）

```python
# llm.py
from abc import ABC, abstractmethod
from typing import Dict, Any, List

class LLMModel(ABC):
    """LLM模型基类"""

    @abstractmethod
    def classify(self, prompt: str, options: List[str]) -> Dict[str, Any]:
        """分类任务"""
        pass

    @abstractmethod
    def extract(self, prompt: str, schema: Dict[str, str]) -> Dict[str, Any]:
        """信息提取"""
        pass

# 实现示例
class OpenAILLM(LLMModel):
    def __init__(self, api_key: str, model: str = "gpt-4"):
        self.api_key = api_key
        self.model = model

    def classify(self, prompt: str, options: List[str]) -> Dict[str, Any]:
        import openai

        response = openai.ChatCompletion.create(
            model=self.model,
            messages=[{"role": "user", "content": prompt}],
            functions=[{
                "name": "classify",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "label": {"type": "string", "enum": options},
                        "confidence": {"type": "number"}
                    }
                }
            }],
            function_call={"name": "classify"}
        )
        # 解析响应...
        return {"label": "single", "confidence": 0.9}

    def extract(self, prompt: str, schema: Dict[str, str]) -> Dict[str, Any]:
        # 类似实现...
        return {}

class MockLLM(LLMModel):
    """Mock LLM用于测试"""
    def classify(self, prompt: str, options: List[str]) -> Dict[str, Any]:
        return {"label": options[0], "confidence": 1.0}

    def extract(self, prompt: str, schema: Dict[str, str]) -> Dict[str, Any]:
        return {k: False for k in schema.keys()}
```

## 3. 使用示例

### 3.1 基本使用

```python
# main.py
from annotator import Annotator
from ocr import PaddleOCR
from llm import OpenAILLM

# 初始化
annotator = Annotator(
    ocr_model=PaddleOCR(),
    llm_model=OpenAILLM(api_key="your-api-key")
)

# 一行代码完成标注
doc = annotator.annotate("document.pdf")

# 输出结果
print(doc.to_dict())
# {
#     "doc_id": "document",
#     "file_type": "pdf",
#     "layout": "single",
#     "has_image": true,
#     "has_table": true,
#     ...
# }
```

### 3.2 分步使用（调试场景）

```python
# 只解析文档
doc = annotator.parse("document.pdf")

# 只做元素检测
doc = annotator.detect(doc)

# 单独测试分类

```

## 6. 总结

### v2核心优势

1. **极简架构**：5个文件完成所有功能
2. **单一入口**：`annotator.annotate()` 一行调用
3. **扁平数据**：Document类无嵌套，直观易懂
4. **灵活调试**：可分步执行任意阶段
5. **易于扩展**：新增功能直接修改对应方法

### v2适用场景

- 快速原型开发
- 中小型项目
- 团队规模小
- 需求相对稳定

### 何时选择v1

- 需要动态配置处理器组合
- 多人协作开发
- 需要处理器级别的复用
- 流程复杂，需要更多控制
